google-cloud-speech>=2.21.0

# silence_cut_app / Auto Cut (standalone desktop app): 2-camera audio sync
# via cross-correlation, plus the customtkinter-based dark GUI. tkinter
# itself ships with the standard python.org macOS installer, but Homebrew's
# python needs `brew install python-tk` alongside it.
numpy>=1.24
scipy>=1.10
customtkinter>=5.2

# create_text_app / Create Text (standalone desktop app): local speech-to-text.
# CPU-only by default; the first run of a given model size downloads its
# weights from Hugging Face Hub (needs network once, then works offline).
faster-whisper>=1.0

# create_text_app manual dictation mode: mic recording for per-segment
# voice input (PortAudio via sounddevice). On Mac this needs no extra
# system install -- the PyPI wheel bundles PortAudio.
sounddevice>=0.4

# create_text_app manual mode's frame preview + customtkinter's own image
# widgets both need Pillow. customtkinter pulls it in transitively, but
# declaring it directly here means `pip install -r requirements.txt` always
# gets it even if that transitive resolution ever changes -- and PyInstaller
# needs it actually installed to bundle it (a "ModuleNotFoundError: No
# module named 'PIL'" in the packaged .app means this wasn't installed when
# `pyinstaller` ran).
Pillow>=10.0

# create_text_app manual mode: real-time video playback (not just scrub
# frames) needs fast, repeated frame reads -- an ffmpeg subprocess per
# frame is far too slow for ~25-30 reads/second. opencv-python-headless
# (no GUI/Qt bindings needed, we draw frames ourselves) keeps one decoder
# open and seeks/reads directly.
opencv-python-headless>=4.8
